cmake_minimum_required(VERSION 3.16)
project(tileplace_array_transactions LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(selection_test selection_test.cpp)
target_compile_options(selection_test PRIVATE -Wall -Wextra -Werror)
add_executable(identity_test identity_test.cpp)
target_compile_options(identity_test PRIVATE -Wall -Wextra -Werror)
add_executable(identity_selection_test identity_selection_test.cpp)
target_compile_options(identity_selection_test PRIVATE -Wall -Wextra -Werror)
add_executable(encoder_test encoder_test.cpp)
target_compile_options(encoder_test PRIVATE -Wall -Wextra -Werror)

add_executable(transaction_search_query transaction_search_query.cpp)
