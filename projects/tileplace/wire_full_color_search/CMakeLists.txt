cmake_minimum_required(VERSION 3.16)
project(tileplace_wire_full_color_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(transaction_native_server transaction_native_server.cpp)
add_executable(transaction_search_query transaction_search_query.cpp)
set(TILEPLACE_HLS_INCLUDE_DIR "" CACHE PATH "Vitis HLS include directory for native builds of staged kernels")
if(TILEPLACE_HLS_INCLUDE_DIR)
    if(NOT EXISTS "${TILEPLACE_HLS_INCLUDE_DIR}/ap_utils.h")
        message(FATAL_ERROR "TILEPLACE_HLS_INCLUDE_DIR must contain ap_utils.h")
    endif()
    foreach(target transaction_native_server transaction_search_query)
        target_include_directories(${target} SYSTEM PRIVATE "${TILEPLACE_HLS_INCLUDE_DIR}")
        target_compile_options(${target} PRIVATE
            "$<$<CXX_COMPILER_ID:GNU,Clang>:-Wno-unknown-pragmas;-Wno-unused-label>")
    endforeach()
endif()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME wire_full_color_search_integration COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p "test_wireenum*.py" -v)
set_tests_properties(wire_full_color_search_integration PROPERTIES
    ENVIRONMENT "TILEPLACE_HLS_INCLUDE_DIR=${TILEPLACE_HLS_INCLUDE_DIR}")
