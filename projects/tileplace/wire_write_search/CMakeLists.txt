cmake_minimum_required(VERSION 3.16)
project(tileplace_wire_write_search LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(transaction_native_server transaction_native_server.cpp)
add_executable(transaction_search_query transaction_search_query.cpp)

enable_testing()
add_test(NAME wire_write_search_integration COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p "test_wirewrite*.py" -v)
