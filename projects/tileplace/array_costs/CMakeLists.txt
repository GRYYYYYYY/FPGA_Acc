cmake_minimum_required(VERSION 3.16)
project(tileplace_array_costs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(score_array_test score_array_test.cpp)
target_compile_options(score_array_test PRIVATE -Wall -Wextra -Werror)
add_executable(scored_native_server scored_native_server.cpp)
add_executable(scored_search_query scored_search_query.cpp)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(net_cost_array_test net_cost_array_test.cpp)
target_compile_options(net_cost_array_test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME array_net_cost_and_overlay COMMAND net_cost_array_test)
add_test(NAME scored_search_integration COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p "test_scored*.py" -v)
add_test(NAME array_score_floor_and_overflow COMMAND score_array_test)

add_executable(cost_pair_array_test cost_pair_array_test.cpp)
target_compile_options(cost_pair_array_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME array_cost_pair_and_routed COMMAND cost_pair_array_test)

add_executable(cost_native_server cost_native_server.cpp)
add_executable(cost_search_query cost_search_query.cpp)
add_test(NAME array_cost_search_integration COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR} -p "test_cost*.py" -v)
